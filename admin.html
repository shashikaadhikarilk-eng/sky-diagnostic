function formatTicketNumber(val, rowIndex) { // Row 2 (පළමු දත්ත පේළිය) Ticket # 1005 බවට පත්කිරීමට (1004 + 1) let defaultNum = `SKY-TICK-${1004 + (parseInt(rowIndex) - 1 || 1)}`; if (!val) return defaultNum; let s = val.toString().trim(); // Date string එකක් අඩංගු වී ඇත්නම් 1005+ අගයට හරවයි if (s.length > 15 || s.includes('Oct') || s.includes('Sep') || s.includes('2026') || s.includes('Mon') || s.includes('GMT') || s.includes('India')) { return defaultNum; } // කුඩා අංක (උදා: 1, 2, 3...) 1004+ එකතු කර 1005, 1006... බවට හරවයි if (s.startsWith('SKY-')) { let numPart = s.replace(/[^0-9]/g, ''); if (numPart && parseInt(numPart) < 1000) { return `SKY-TICK-${1004 + parseInt(numPart)}`; } return s.startsWith('SKY-TICK-') ? s : s.replace('SKY-', 'SKY-TICK-'); } if (!isNaN(s) && parseInt(s) < 1000) { return `SKY-TICK-${1004 + parseInt(s)}`; } return s; }
